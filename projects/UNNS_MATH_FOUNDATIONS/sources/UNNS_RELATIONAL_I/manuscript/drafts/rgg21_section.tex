\section{Setting}
By Corollary~10.2, an integer nest output $M\perp N$ has $N\mid M$; writing $M=kN$, $d=N+1\ge 2$ gives value $v(r)=kd^2$ for the \emph{anchor route} $r=(k,d)$. For $m\ge1$ let $F(m)=\{r: v(r)=m\}$ (the \emph{fibre}); $|F(m)|=R(m)$. Field occurrences ($N\nmid M$) are outside this section.

\begin{definition}[Joint continuation $\vj$; proposed]
$r\vj s=(\lcm(k_r,k_s),\ \lcm(d_r,d_s))$. Route order: $r\le s$ iff $k_r\mid k_s$ and $d_r\mid d_s$.
\end{definition}

\begin{lemma}[Order structure; proved]\label{lem:order}
$\vj$ is the least upper bound for $\le$ (hence commutative, associative, idempotent); $v(r)\mid v(r\vj s)$ and $v(s)\mid v(r\vj s)$; $v$ is strictly increasing in each coordinate, so every fibre is an antichain.
\end{lemma}
\begin{proof}
Lub: componentwise lcm. Divisibility of values is immediate from $kd^2$. If $r\le r'$ with $v(r)=v(r')$ then $k'/k\cdot (d'/d)^2=1$ with both ratios positive integers, so $r=r'$.
\end{proof}

\section{Fibre separation and relational distance}
\begin{theorem}[Join value; proved]\label{thm:join}
For $r=(m/d^2,d)$, $r'=(m/d'^2,d')\in F(m)$ put $g=\gcd(d,d')$, $L=\lcm(d,d')$. Then $\lcm(k,k')=m/g^2$ and
\[ v(r\vj r') = m\Bigl(\tfrac{L}{g}\Bigr)^{2}. \]
\end{theorem}
\begin{proof}
For a prime $p$ with $\nu_p(m)=e$, $\nu_p(d)=a$, $\nu_p(d')=a'$: $\nu_p\lcm(k,k')=\max(e-2a,e-2a')=e-2\min(a,a')=\nu_p(m/g^2)$. Multiply by $L^2$.
\end{proof}
\begin{corollary}[Universal pairwise separation; proved]\label{cor:sep}
If $r\ne r'$ in $F(m)$ then $v(r\vj r')>m=v(r\vj r)$. Hence the joint-continuation value with probe $s=r$ distinguishes $r$ from every other route of its fibre. (This is a consequence of the definition of $\vj$, not independent evidence of interaction.)
\end{corollary}
\begin{proof}
$r\neq r'$ forces $d\ne d'$ (as $k=m/d^2$), so $L>g$.
\end{proof}

\begin{definition}[Relational distance]
$D(r,r')=\log\bigl(v(r\vj r')/m\bigr)=2\log(L/g)=2\sum_p|\nu_p(d)-\nu_p(d')|\log p$ for $r,r'\in F(m)$.
\end{definition}
\begin{theorem}[Fibre geometry; proved]\label{thm:geom}
$D$ is a metric on $F(m)$. With $b_p=\lfloor \nu_p(m)/2\rfloor$, $F(m)$ is isometric to $\prod_p\{0,\dots,b_p\}\setminus\{0\}$ with weighted $\ell^1$ metric (weights $2\log p$); so $|F(m)|=\prod_p(b_p+1)-1=R(m)$. Its diameter is $2\sum_p b_p\log p$ if at least two $b_p\ge1$, and $2(b_p-1)\log p$ if exactly one.
\end{theorem}
\begin{proof}
$d\mapsto(\nu_p(d))_p$ identifies $F(m)$ with exponent vectors $0\le a_p\le b_p$ other than $0$ (the condition $d^2\mid m$); the formula for $D$ is the weighted $\ell^1$ distance. Diameter: the full box has diameter $2\sum b_p\log p$ attained by opposite corners; if at least two primes have $b_p\ge1$, split them into two nonempty groups: the point equal to $b$ on the first group and $0$ elsewhere, and the point equal to $b$ on the second group and $0$ elsewhere, are both nonzero and at full distance; if only one, the origin removal shortens the chain by one step.
\end{proof}

\section{Refinement geometry}
For a prime $q$ put $T_q(k,d)=(k/q^2,dq)$ when $q^2\mid k$. It preserves $v$; its inverse is $(k,d)\mapsto(kq^2,d/q)$ when $q\mid d$ and $d/q\ge2$.
\begin{theorem}[Joint covariance; proved]\label{thm:cov}
If $T_q r,T_q s$ are defined, $T_qr\vj T_qs=T_q(r\vj s)$. Consequently $v$ of the joint continuation, and $D$, are invariant under joint refinement, while one refinement step moves a route by $D(T_qr,r)=2\log q$. Invariance under refinement of \emph{one} argument fails: $v(T_qr\vj s)\ne v(r\vj s)$ in $5098$ of $10304$ tested cases (checked).
\end{theorem}
\begin{proof}
$\lcm(k/q^2,k'/q^2)=\lcm(k,k')/q^2$ and $\lcm(dq,d'q)=q\lcm(d,d')$.
\end{proof}
\begin{theorem}[Distance equals refinement cost; proved]\label{thm:geo}
Give each refinement step on prime $q$ the cost $2\log q$. Then for $r,r'\in F(m)$ the minimal total cost of a chain of refinement steps (and inverses) from $r$ to $r'$ equals $D(r,r')$.
\end{theorem}
\begin{proof}
A step changes one coordinate $a_p$ by $\pm1$, so cost $\ge\sum_p|a_p-a'_p|\,2\log p=D$. Conversely, the vector $\max(a,a')$ is $\le b$ coordinatewise and nonzero, hence a fibre route (it is $d=\lcm(d,d')$); the path $a\to\max(a,a')\to a'$ stays in the fibre (never at $d=1$) and costs exactly $D$.
\end{proof}

\section{A set-theoretic quotient criterion (no bilinearity)}
Let $\Pi$ be an observation, $\sim_\Pi$ the induced equivalence, and $J$ a (possibly set-valued) joint operation; $\Pi_*J(r,s)$ denotes the image set. Say $J$ \emph{descends} to $\Pi$ if $r\sim_\Pi r'\Rightarrow \Pi_*J(r,s)=\Pi_*J(r',s)$ and likewise in the second slot.
\begin{theorem}[Set-theoretic Quotient Criterion; proved]\label{thm:setq}
Let $\mathcal T$ be a family of refinements and $\sim_{\mathcal T}$ the equivalence generated by $\{(r,Tr)\}$. Assume $\sim_{\mathcal T}=\sim_\Pi$. Then $J$ descends to $\Pi$ iff $\Pi_*J(Tr,s)=\Pi_*J(r,s)$ and $\Pi_*J(r,Ts)=\Pi_*J(r,s)$ for all generating $T,r,s$.
\end{theorem}
\begin{proof}
For fixed $s$ the relation $\{(r,r'):\Pi_*J(r,s)=\Pi_*J(r',s)\}$ is an equivalence relation; it contains $\sim_{\mathcal T}$ iff it contains the generators. Same for the second slot.
\end{proof}
\begin{corollary}[Application; proved/checked]
Hypothesis $\sim_{\mathcal T}=\sim_{\rm value}$ holds for anchor routes (Theorem~\ref{thm:geo}: fibres are connected by prime refinements). The product $J_\times(r,s)=(k_rk_s,d_rd_s)$ satisfies value-invariance and descends (checked); $\vj$ violates one-argument invariance and does not descend.
\end{corollary}
\begin{definition}[Set deficit]
Let $\kappa(m)$ be the number of classes of the largest $J$-congruence inside $\sim_\Pi$ on $F(m)$, and $\delta_{\rm set}(m)=\kappa(m)-1$. Then $\delta_{\rm set}(m)=R(m)-1$ for $\vj$ (Corollary~\ref{cor:sep}) and $0$ for $J_\times$ with value observation.
\end{definition}

\section{Single-probe fibre reconstruction}
Let $h(m)=\prod_pp^{\lfloor\nu_p(m)/2\rfloor}$ and $s_m=(m/h(m)^2,\ h(m))$.
\begin{theorem}[Single-Probe Fibre Reconstruction; proved]\label{thm:single}
$F(m)\ne\varnothing$ iff $m$ is not squarefree iff $h(m)\ge2$, and then $s_m\in F(m)$. For every $r=(m/d^2,d)\in F(m)$,
\[ r\vj s_m=\bigl(m/d^2,\ h(m)\bigr),\qquad v(r\vj s_m)=m\Bigl(\tfrac{h(m)}{d}\Bigr)^{2}. \]
Hence $d=h(m)\sqrt{m/v(r\vj s_m)}$ and the entire route $r$ is recovered from one joint-continuation value with the fixed probe $s_m$.
\end{theorem}
\begin{proof}
$d^2\mid m$ implies $\nu_p(d)\le\lfloor\nu_p(m)/2\rfloor$ for every $p$, so $d\mid h$; in particular $h^2\mid m$ and $s_m\in F(m)$ when $h\ge2$. Then $\lcm(d,h)=h$ and $m/h^2\mid m/d^2$ gives $\lcm(m/d^2,m/h^2)=m/d^2$. The value is $(m/d^2)h^2$.
\end{proof}
\begin{corollary}[Probe family; proved]
Any probe $s=(k_s,d_s)$ with $h(m)\mid d_s$ and $k_s\mid m/h(m)^2$ resolves $F(m)$: $\lcm(d,d_s)=d_s$ for all $d\mid h$, and $\lcm(m/d^2,k_s)=m/d^2$, so $v(r\vj s)=(m/d^2)d_s^2$ is injective in $d$. (Sufficient, not claimed necessary.) Probes need not lie in the fibre.
\end{corollary}
\begin{remark}
$s_m$ depends on $m$, so the probe is fixed per fibre, not universal across fibres. Checked: all $7840$ nonempty fibres with $m\le 20000$; fibre of $72$: $h=6$, $s_{72}=(2,6)$, join values $648,288,72$ for $(18,2),(8,3),(2,6)$.
\end{remark}

\section{Robustness under coordinate choice, and an erratum}
\begin{lemma}[Separation is coordinate-robust; proved]\label{lem:robust}
Let routes be coordinates in a poset chart with a value $v$ strictly increasing in each coordinate, and let $J$ satisfy $J(r,s)\ge r,s$ coordinatewise. Then for distinct $r,r'$ with equal value, $v(J(r,r'))>v(r)$.
\end{lemma}
\begin{proof}
If $J(r,r')=r$ then $r'\le r$; with equal value strict monotonicity forces $r'=r$. So $J(r,r')\ge r$, $\ne r$, and strict monotonicity gives $v(J(r,r'))>v(r)$.
\end{proof}
This applies to joins in $(k,d)$, in $(k,N)$, and in occurrence coordinates $(M,N)$ (value $M(N+1)^2/N$ is strictly increasing in $M$ and in $N\ge1$; the occurrence join is anchor-closed because $\lcm(N,N')\mid\lcm(M,M')$). Checked on $2745$ same-value pairs ($m\le6000$). The exact formulas of Theorems~\ref{thm:join}--\ref{thm:geo} are specific to $(k,d)$ coordinates; that $s_m$ also resolves every fibre under the other two joins is checked ($693/693$), not proved.

\paragraph{Erratum (RGG-2 note).} The statement ``the mixed operations stay in the fibre, so the fibre is a lattice'' is false on the anchor domain $d\ge2$. $(\lcm k,\gcd d)$ returns $(m/g^2,g)$, which is inadmissible when $g=\gcd(d,d')=1$: e.g.\ $(18,2),(8,3)\mapsto(72,1)$ ($501$ of $2745$ same-value pairs). $(\gcd k,\lcm d)=(m/L^2,L)$ is always admissible and is the least upper bound in the $d$-divisibility order. Hence $F(m)$ is a \emph{join-semilattice}, not a lattice (pairs with $\gcd(d,d')=1$ have no common lower bound); it becomes a lattice when the formal route $(m,1)$ is adjoined (checked, as it is the full grid $\prod\{0..b_p\}$).
